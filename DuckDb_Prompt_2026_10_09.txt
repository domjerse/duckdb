-- Tabela Ficticia 1Milhão
DROP TABLE IF EXISTS tb_milhao; CREATE OR REPLACE TEMP TABLE tb_milhao AS
SELECT
    i AS id,
    upper(substr(md5(random()::VARCHAR), 1, 4)) || '-' ||  lpad(i::VARCHAR, 6, '0') AS codigo,
    CASE WHEN random() < 0.8 THEN 'ATIVO' ELSE 'INATIVO' END AS status,
    CASE (i % 5) WHEN 0 THEN 'RJ' WHEN 1 THEN 'SP' WHEN 2 THEN 'MG' WHEN 3 THEN 'BA' ELSE 'PR' END AS uf,
    CASE (i % 4)
        WHEN 0 THEN 'Direito'
        WHEN 1 THEN 'Engenharia'
        WHEN 2 THEN 'Administração'
        ELSE 'Medicina'
    END AS curso,
    ROUND(100 + random() * 400, 2) AS valor, -- Valor entre 100 e 400
    CASE CAST(floor(random() * 4) AS INTEGER) WHEN 0 THEN 30 WHEN 1 THEN 40 WHEN 2 THEN 50 ELSE 100 END AS desconto,
    DATE '2025-01-01' + CAST(i % 730 AS INTEGER) AS data_ref
FROM range(1,1000001) t(i);

-- #### Processo de backup e restauração dos notebooks da DuckDB UI
-- 1. Exportar a versão mais recente de cada notebook
COPY (  SELECT * FROM _duckdb_ui.main.notebook_versions
  	  QUALIFY ROW_NUMBER() OVER (PARTITION BY notebook_id
        ORDER BY version DESC, created DESC ) = 1
) TO 'C:/DuckDB/backup/notebooks_latest.json' FORMAT JSON, ARRAY true);

-- 2. Exporta uma versão por notebook, escolhendo a de maior número de versão.
COPY (
    SELECT * FROM _duckdb_ui.main.notebooks
)
TO 'C:/DuckDB/backup/notebooks.json' (FORMAT JSON, ARRAY true);

-- 3. Importar as versões posteriormente
INSERT INTO _duckdb_ui.main.notebook_versions
SELECT  notebook_id::UUID, version::INTEGER, title::VARCHAR,
    json::VARCHAR, created::TIMESTAMP, expires::TIMESTAMP
FROM read_json_auto( 'C:/DuckDB/backup/notebooks_latest.json' );
